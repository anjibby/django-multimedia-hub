class Student:
    def __init__(self, name, age, major):
        self.name = name
        self.age = age
        self.major = major  # New attribute

    def introduce(self):
        print(f"Hi, my name is {self.name} and I am {self.age} years old.")

    # New method
    def study(self):
        print(f"{self.name} is studying {self.major}.")

# Create an object
user = Student("Anji", 17, "Physiology")

# Call both methods
user.introduce()
user.study()
class Student:
    def __init__(self, name, age, major):
        self.name = name
        self.age = age
        self.major = major

    def introduce(self):
        print(f"Hi, my name is {self.name} and I am {self.age} years old.")

    def study(self):
        print(f"{self.name} is studying {self.major}.")

# Child Class inheriting from Student
class Prefect(Student):
    def __init__(self, name, age, major, role):
        # Call the parent class (Student) constructor
        super().__init__(name, age, major)
        self.role = role  # Unique attribute for Prefect

    # Unique method for Prefect
    def assign_duty(self):
        print(f"{self.name} is the {self.role} and is maintaining order.")

# Create a regular student
student1 = Student("Alex", 18, "Biology")
student1.introduce()

print("---")

# Create a prefect
prefect1 = Prefect("Anji", 17, "Physiology", "Health Prefect")
prefect1.introduce()    # Inherited from Student
prefect1.study()        # Inherited from Student
prefect1.assign_duty()  # Unique to Prefect
class BankAccount:
    def __init__(self, balance):
        self.__balance = balance  # Private attribute

    def deposit(self, amount):
        if amount > 0:
            self.__balance += amount

    def get_balance(self):
        return self.__balance

account = BankAccount(100)
# account.__balance  <-- This will throw an error!
print(account.get_balance()) # Output: 100
